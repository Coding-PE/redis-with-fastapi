# Redis Sorted Set 명령어 모음

> 강의노트 Ch.03 — Redis 기본 자료형과 명령어 / Sorted Set

---

## 1. 기본 명령어

```bash
# ZADD — 멤버·Score 추가
127.0.0.1:6379> ZADD ranking:game 9500 "Kim" 8200 "Lee" 7100 "Park"
# (integer) 3   ← 새로 추가된 멤버 수

# 이미 존재하는 멤버의 Score 업데이트
127.0.0.1:6379> ZADD ranking:game 9800 "Kim"
# (integer) 0   ← 새로 추가된 멤버 없음 (Score만 업데이트)

# ZREM — 멤버 삭제
127.0.0.1:6379> ZREM ranking:game "Park"
# (integer) 1   ← 삭제된 멤버 수

# ZSCORE — Score 조회
127.0.0.1:6379> ZSCORE ranking:game "Kim"
# "9800"

# 존재하지 않는 멤버 조회
127.0.0.1:6379> ZSCORE ranking:game "Choi"
# (nil)

# ZINCRBY — Score 증감
127.0.0.1:6379> ZINCRBY ranking:game 500 "Lee"
# "8700"   ← 기존 8200 + 500

# ZRANK — 오름차순 순위 조회 (0부터 시작)
127.0.0.1:6379> ZRANK ranking:game "Lee"
# (integer) 0   ← Score가 낮아서 인덱스 0

127.0.0.1:6379> ZRANK ranking:game "Kim"
# (integer) 1   ← Score가 높아서 인덱스 1

# ZREVRANK — 내림차순 순위 조회 (0부터 시작)
127.0.0.1:6379> ZREVRANK ranking:game "Kim"
# (integer) 0   ← Score가 가장 높아서 인덱스 0

# Score와 함께 조회 (Redis 7.2+)
127.0.0.1:6379> ZREVRANK ranking:game "Kim" WITHSCORE
# 1) (integer) 0
# 2) "9800"

# ZCARD — 멤버 수 확인
127.0.0.1:6379> ZCARD ranking:game
# (integer) 2
```

> ⚠️ 순위는 **0부터 시작**합니다. 실무에서 "1위"를 표시하려면 애플리케이션에서 +1 처리가 필요합니다.

## 2. 범위 조회 명령어

```bash
# 실습 데이터 준비
127.0.0.1:6379> ZADD ranking:game 9800 "Kim" 8700 "Lee" 7100 "Park" 6500 "Choi"

# ZRANGE — 인덱스 기반 전체 조회 (오름차순)
127.0.0.1:6379> ZRANGE ranking:game 0 -1
# 1) "Choi"
# 2) "Park"
# 3) "Lee"
# 4) "Kim"

# Score 포함 전체 조회
127.0.0.1:6379> ZRANGE ranking:game 0 -1 WITHSCORES
# 1) "Choi"
# 2) "6500"
# 3) "Park"
# 4) "7100"
# 5) "Lee"
# 6) "8700"
# 7) "Kim"
# 8) "9800"

# 내림차순 조회 (REV)
127.0.0.1:6379> ZRANGE ranking:game 0 -1 REV WITHSCORES
# 1) "Kim"
# 2) "9800"
# 3) "Lee"
# 4) "8700"
# 5) "Park"
# 6) "7100"
# 7) "Choi"
# 8) "6500"

# Score 기반 범위 조회 (BYSCORE)
127.0.0.1:6379> ZRANGE ranking:game 7000 9000 BYSCORE WITHSCORES
# 1) "Park"
# 2) "7100"
# 3) "Lee"
# 4) "8700"

# -inf ~ +inf = 전체 조회
127.0.0.1:6379> ZRANGE ranking:game -inf +inf BYSCORE
# 1) "Choi"
# 2) "Park"
# 3) "Lee"
# 4) "Kim"

# 상위 3명만 조회 (내림차순 + LIMIT)
127.0.0.1:6379> ZRANGE ranking:game +inf -inf BYSCORE REV LIMIT 0 3 WITHSCORES
# 1) "Kim"
# 2) "9800"
# 3) "Lee"
# 4) "8700"
# 5) "Park"
# 6) "7100"

# ZCOUNT — Score 범위 내 멤버 수
127.0.0.1:6379> ZCOUNT ranking:game 7000 9000
# (integer) 2   ← Park(7100), Lee(8700)

127.0.0.1:6379> ZCOUNT ranking:game -inf +inf
# (integer) 4

# ZPOPMIN — Score 가장 낮은 멤버 꺼내기
127.0.0.1:6379> ZPOPMIN ranking:game
# 1) "Choi"
# 2) "6500"

# ZPOPMAX — Score 가장 높은 멤버 2개 꺼내기
127.0.0.1:6379> ZPOPMAX ranking:game 2
# 1) "Kim"
# 2) "9800"
# 3) "Lee"
# 4) "8700"

# ZREMRANGEBYSCORE — Score 범위 내 멤버 삭제
127.0.0.1:6379> ZREMRANGEBYSCORE ranking:game -inf 6999
# (integer) 1   ← 삭제된 멤버 수 (Choi 삭제)
```

## 3. ZSCAN — 대량 멤버 조회

```bash
# 커서 0부터 시작
127.0.0.1:6379> ZSCAN ranking:game 0
# 1) "0"
# 2) 1) "Park"
#    2) "7100"
#    3) "Lee"
#    4) "8700"
#    5) "Kim"
#    6) "9800"
```

## 4. 실무 활용 예시

```bash
# 실시간 랭킹 (Leaderboard)
127.0.0.1:6379> ZADD ranking:game 9800 "Kim" 8700 "Lee" 7100 "Park"

# 점수 업데이트
127.0.0.1:6379> ZINCRBY ranking:game 500 "Lee"

# 상위 3명 조회 (내림차순)
127.0.0.1:6379> ZRANGE ranking:game 0 2 REV WITHSCORES

# 특정 유저 순위 확인 (0 = 1등)
127.0.0.1:6379> ZREVRANK ranking:game "Kim"

# 지연 작업 큐 (Score = 실행할 Unix timestamp)
127.0.0.1:6379> ZADD delayed:jobs 1735689600 "send_email:user:1"
127.0.0.1:6379> ZADD delayed:jobs 1735693200 "push_notification:user:2"

# 지금 실행해야 할 작업 1개 조회 (현재 시간 이하)
127.0.0.1:6379> ZRANGE delayed:jobs -inf 1735689600 BYSCORE LIMIT 0 1

# 처리 후 삭제
127.0.0.1:6379> ZREM delayed:jobs "send_email:user:1"

# 선착순 대기열 (Score = 요청 timestamp)
127.0.0.1:6379> ZADD waiting:queue 1735689600001 "user:1"
127.0.0.1:6379> ZADD waiting:queue 1735689600002 "user:2"
127.0.0.1:6379> ZADD waiting:queue 1735689600003 "user:3"

# 대기열 순서대로 조회 (오름차순 = 먼저 클릭한 순)
127.0.0.1:6379> ZRANGE waiting:queue 0 -1 WITHSCORES

# 특정 유저의 대기 순위 확인 (0 = 1번째)
127.0.0.1:6379> ZRANK waiting:queue "user:2"
# (integer) 1   ← 2번째 대기 중

# 가장 먼저 클릭한 사람 빼오기 (처리 후 자동 삭제)
127.0.0.1:6379> ZPOPMIN waiting:queue
# 1) "user:1"
# 2) "1735689600001"
```
