## 실습 코드

```python
import time
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import redis.asyncio as redis

@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url("redis://localhost:6379/0", decode_responses=True)
    yield
    await app.state.redis.aclose()

app = FastAPI(lifespan=lifespan)

# 제한 설정: 1분당 최대 5회 (시연을 위해 짧게 설정)
LIMIT = 5
WINDOW = 60  # 초

@app.middleware("http")
async def rate_limit_middleware(request: Request, call_next):
    # Swagger UI 접속은 제한에서 제외
    if request.url.path in ["/docs", "/openapi.json"]:
        return await call_next(request)

    rd = request.app.state.redis

    # Rate Limiting 기준 식별자 (IP 기준)
    # 서비스 정책에 따라 user_id, api_key 등으로 변경 가능
    user_identifier = request.client.host if request.client else "127.0.0.1"

    # 현재 시간을 WINDOW(60초) 단위의 정수로 변환
    # 같은 60초 안의 요청은 동일한 current_minute 값을 가짐
    current_minute = int(time.time() // WINDOW)
    cache_key = f"rate_limit:{user_identifier}:{current_minute}"

    # 1. 카운트 증가 (키가 없으면 자동 생성 후 1 반환)
    count = await rd.incr(cache_key)

    # 2. 첫 요청(count == 1)일 때만 만료 시간 설정
    if count == 1:
        await rd.expire(cache_key, WINDOW)

    # 3. 제한 초과 여부 확인
    if count > LIMIT:
        return JSONResponse(
            status_code=429,
            content={
                "error": "Too Many Requests",
                "detail": f"1분에 {LIMIT}회까지만 요청 가능합니다.",
                "retry_after": f"{WINDOW - (int(time.time()) % WINDOW)}s"
            }
        )

    # 4. 제한 통과 → 실제 API 핸들러로 요청 전달
    response = await call_next(request)
    return response

@app.get("/data")
async def get_sensitive_data():
    return {"data": "This is protected by rate limiting"}
```

## 테스트 클라이언트

`test_rate_limit.py` 파일 생성:

```python
# test_rate_limit.py — 10개 요청을 연속 전송
import asyncio
import httpx

async def send_request(client, req_num):
    response = await client.get("http://localhost:8000/data", timeout=5.0)
    print(f"요청 {req_num:2d} → {response.status_code} {response.json()}")

async def main():
    async with httpx.AsyncClient() as client:
        for i in range(1, 11):
            await send_request(client, i)

asyncio.run(main())
```

```bash
python test_rate_limit.py
```
