## 실습 코드

```python
import uuid
from contextlib import asynccontextmanager
from fastapi import FastAPI, Response, Cookie, Request
from pydantic import BaseModel
import redis.asyncio as redis

@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url("redis://localhost:6379/0", decode_responses=True)
    yield
    await app.state.redis.aclose()

app = FastAPI(lifespan=lifespan)

SESSION_EXPIRE = 3600  # 세션 만료 시간: 1시간 (초)

# 로그인 요청 데이터 구조 정의 (Java의 DTO와 동일한 역할)
class LoginRequest(BaseModel):
    user_id: str

@app.post("/login")
async def login(req_data: LoginRequest, response: Response, request: Request):
    rd = request.app.state.redis

    # 1. 고유하고 추측 불가능한 세션 ID 생성
    session_id = str(uuid.uuid4())
    session_key = f"session:{session_id}"

    # 2. 유저 정보를 Hash로 저장
    user_info = {
        "user_id": req_data.user_id,
        "tier": "Premium",
        "ip": request.client.host if request.client else "127.0.0.1"
    }
    await rd.hset(session_key, mapping=user_info)

    # 3. 세션 만료 시간 설정 (1시간)
    await rd.expire(session_key, SESSION_EXPIRE)

    # 4. 클라이언트 쿠키에 세션 ID 전달
    response.set_cookie(
        key="session_id",
        value=session_id,
        httponly=True,       # JS에서 쿠키 접근 차단 (XSS 방어)
        secure=False,        # 운영 환경에서는 반드시 True로 변경
        samesite="lax"
    )
    return {"message": "Login Success", "session_id": session_id}

@app.get("/me")
async def get_my_info(request: Request, session_id: str | None = Cookie(None)):
    if not session_id:
        return {"error": "Not logged in"}

    rd = request.app.state.redis
    session_key = f"session:{session_id}"

    # 1. Redis에서 세션 정보 조회
    user_info = await rd.hgetall(session_key)
    if not user_info:
        return {"error": "Session expired or invalid"}

    # 2. 활동 중이므로 세션 만료 시간 연장 (Sliding Window 방식)
    await rd.expire(session_key, SESSION_EXPIRE)

    return user_info

@app.post("/logout")
async def logout(response: Response, request: Request, session_id: str | None = Cookie(None)):
    if session_id:
        rd = request.app.state.redis
        # Redis에서 세션 즉시 삭제
        await rd.delete(f"session:{session_id}")
        # 클라이언트 쿠키도 삭제
        response.delete_cookie("session_id")
    return {"message": "Logged out"}
```

> ⚠️ `secure=False`는 개발 환경 전용입니다. 운영 환경 배포 시 반드시 `secure=True`로 변경하세요.
