## 실습 코드

```python
import time
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request, HTTPException
import redis.asyncio as redis

@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url("redis://localhost:6379/0", decode_responses=True)
    yield
    await app.state.redis.aclose()

app = FastAPI(lifespan=lifespan)

QUEUE_KEY = "queue:train:2026-09-27"  # 명절 기차표 예매 대기열

@app.post("/queue/enter")
async def enter_queue(user_id: str, request: Request):
    """대기열에 입장합니다. 접속 시각이 곧 순번입니다."""
    rd = request.app.state.redis

    # ZADD NX: 이미 대기열에 있으면 score를 갱신하지 않음 (순번 유지)
    # score = 접속 시각 (Unix 타임스탬프) → 먼저 접속할수록 낮은 score → 앞 순번
    joined = await rd.zadd(QUEUE_KEY, {user_id: time.time()}, nx=True)

    rank = await rd.zrank(QUEUE_KEY, user_id)
    total = await rd.zcard(QUEUE_KEY)

    if joined == 0:
        # 이미 대기열에 있는 경우 → 현재 순번 반환
        return {
            "message": "이미 대기열에 있습니다.",
            "my_rank": rank + 1,
            "total_waiting": total
        }

    return {
        "message": "대기열 입장 완료",
        "my_rank": rank + 1,       # 0-indexed → 1-indexed 변환
        "total_waiting": total
    }

@app.get("/queue/status/{user_id}")
async def get_queue_status(user_id: str, request: Request):
    """내 현재 순번과 전체 대기 인원을 조회합니다."""
    rd = request.app.state.redis

    rank = await rd.zrank(QUEUE_KEY, user_id)

    if rank is None:
        raise HTTPException(status_code=404, detail="대기열에 없습니다.")

    total = await rd.zcard(QUEUE_KEY)

    return {
        "user_id": user_id,
        "my_rank": rank + 1,
        "total_waiting": total,
        "message": f"현재 {rank + 1}번째 대기 중입니다. (전체 {total}명)"
    }

@app.post("/queue/admit")
async def admit_next(count: int = 1, request: Request = None):
    """
    선착순으로 N명을 입장 허용합니다.
    실제 서비스에서는 예매 서버가 처리 가능한 속도에 맞춰 주기적으로 호출합니다.
    """
    rd = request.app.state.redis

    # ZPOPMIN: score가 가장 낮은(가장 먼저 접속한) 사람을 꺼냄
    admitted = await rd.zpopmin(QUEUE_KEY, count)

    if not admitted:
        return {"message": "대기열이 비어있습니다."}

    admitted_users = [user_id for user_id, score in admitted]

    return {
        "message": f"{len(admitted_users)}명 입장 허용",
        "admitted_users": admitted_users,
        "remaining": await rd.zcard(QUEUE_KEY)
    }
```

## 테스트 순서

### ① 여러 명 대기열 입장

```
POST http://localhost:8000/queue/enter?user_id=user_a
POST http://localhost:8000/queue/enter?user_id=user_b
POST http://localhost:8000/queue/enter?user_id=user_c
```

### ② 내 순번 조회

```
GET http://localhost:8000/queue/status/user_b
```

### ③ 중복 입장 시도 (순번 유지 확인)

```
POST http://localhost:8000/queue/enter?user_id=user_a
# {"message": "이미 대기열에 있습니다.", "my_rank": 1, "total_waiting": 3}
```

### ④ 선착순 입장 허용

```
POST http://localhost:8000/queue/admit?count=1
# {"message": "1명 입장 허용", "admitted_users": ["user_a"], "remaining": 2}
```
