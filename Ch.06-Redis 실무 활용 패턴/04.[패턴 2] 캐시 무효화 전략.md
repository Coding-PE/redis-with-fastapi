## 2. 이벤트 기반 무효화 (DB 변경 시 캐시 즉시 삭제)

```python
import json
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request
import redis.asyncio as redis

fake_db = {
    "user:1": {"name": "Kim", "email": "kim@example.com", "tier": "Gold"},
    "user:2": {"name": "Lee", "email": "lee@example.com", "tier": "Silver"}
}

@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url("redis://localhost:6379/0", decode_responses=True)
    yield
    await app.state.redis.aclose()

app = FastAPI(lifespan=lifespan)

@app.put("/users/{user_id}")
async def update_user_invalidation(user_id: str, request: Request):
    rd = request.app.state.redis

    # 1. DB 업데이트
    fake_db[f"user:{user_id}"]["email"] = "new@example.com"

    # 2. 캐시 즉시 삭제
    cache_key = f"user:profile:{user_id}"
    await rd.delete(cache_key)

    return {"message": "업데이트 완료, 캐시 삭제됨"}
```

## 3. Write-Through (DB + 캐시 동시 갱신)

```python
@app.put("/users/{user_id}/write-through")
async def update_user_write_through(user_id: str, request: Request):
    rd = request.app.state.redis

    # 1. DB 업데이트
    updated_data = {"name": "Kim", "email": "new@example.com", "tier": "Gold"}
    fake_db[f"user:{user_id}"] = updated_data

    # 2. 캐시도 즉시 최신 데이터로 갱신
    cache_key = f"user:profile:{user_id}"
    await rd.set(cache_key, json.dumps(updated_data), ex=300)

    return {"message": "업데이트 완료, 캐시 갱신됨"}
```

## 4. 버전 기반 무효화

```python
# 버전 정보를 별도 키로 관리
async def get_cache_version(rd, user_id: str) -> str:
    version = await rd.get(f"user:version:{user_id}")
    return version or "1"

@app.get("/users/{user_id}/versioned")
async def get_user_versioned(user_id: str, request: Request):
    rd = request.app.state.redis

    # 현재 버전 조회
    version = await get_cache_version(rd, user_id)
    cache_key = f"user:profile:{user_id}:v{version}"

    cached = await rd.get(cache_key)
    if cached:
        return json.loads(cached)

    user_data = fake_db.get(f"user:{user_id}")
    await rd.set(cache_key, json.dumps(user_data), ex=300)
    return user_data

@app.put("/users/{user_id}/versioned")
async def update_user_versioned(user_id: str, request: Request):
    rd = request.app.state.redis

    # DB 업데이트
    fake_db[f"user:{user_id}"]["email"] = "new@example.com"

    # 버전 증가 (v1 → v2) → 이전 캐시 키는 자동으로 조회되지 않음
    await rd.incr(f"user:version:{user_id}")

    return {"message": "업데이트 완료, 버전 증가됨"}
```

## 5. 캐시 워밍 (Cache Warming)

```python
@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url("redis://localhost:6379/0", decode_responses=True)

    # 서버 시작 시 인기 데이터 미리 캐싱 (캐시 워밍)
    for user_id, data in fake_db.items():
        await app.state.redis.set(f"cache:{user_id}", json.dumps(data), ex=300)

    yield
    await app.state.redis.aclose()
```
