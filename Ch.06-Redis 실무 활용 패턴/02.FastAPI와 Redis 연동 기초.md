## 1. Redis 연결 (lifespan + app.state 패턴)

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI
import redis.asyncio as redis

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 서버 시작 시: Redis 연결 생성 후 app.state에 저장
    app.state.redis = redis.from_url(
        "redis://localhost:6379/0",
        decode_responses=True
    )
    print("✅ Redis 연결 성공!")

    yield

    # 서버 종료 시: Redis 연결 안전하게 해제
    await app.state.redis.aclose()
    print("❌ Redis 연결 해제!")

app = FastAPI(lifespan=lifespan)
```

## 2. SET / GET API

```python
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request
import redis.asyncio as redis

@asynccontextmanager
async def lifespan(app: FastAPI):
    app.state.redis = redis.from_url(
        "redis://localhost:6379/0",
        decode_responses=True
    )
    print("✅ Redis 연결 성공!")
    yield
    await app.state.redis.aclose()
    print("❌ Redis 연결 해제!")

app = FastAPI(lifespan=lifespan)

@app.post("/items/{item_id}")
async def set_item(item_id: str, value: str, request: Request):
    """Redis에 데이터를 저장합니다. (SET)"""
    rd = request.app.state.redis
    key = f"item:{item_id}"

    await rd.set(key, value)

    return {"message": "저장 완료", "key": key, "value": value}

@app.get("/items/{item_id}")
async def get_item(item_id: str, request: Request):
    """Redis에서 데이터를 조회합니다. (GET)"""
    rd = request.app.state.redis
    key = f"item:{item_id}"

    result = await rd.get(key)

    if result is None:
        return {"error": "데이터가 없습니다."}

    return {"key": key, "value": result}
```

## 4. redis-py 기본 명령어 대응표

| Redis CLI | Python (redis-py) |
|-----------|-------------------|
| `SET key value` | `await rd.set('key', 'value')` |
| `GET key` | `await rd.get('key')` |
| `DEL key` | `await rd.delete('key')` |
| `EXPIRE key 60` | `await rd.expire('key', 60)` |
| `SET key value EX 60` | `await rd.set('key', 'value', ex=60)` |
| `INCR key` | `await rd.incr('key')` |

```python
# ❌ 사용하지 마세요 (deprecated)
await rd.setex('key', 60, 'value')

# ✅ 올바른 방식
await rd.set('key', 'value', ex=60)
```
